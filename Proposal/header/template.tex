\makeatletter
\def\makecoverpage{%
\begin{titlepage}
  \thispagestyle{empty}
  \vspace*{3cm}
  \noindent\rule{2pt}{10cm}\hspace{1.5cm}%
  \begin{minipage}{0.7\textwidth}
    {\LARGE\bfseries $if(title)$$title$$else$Project Title$endif$\par}

    $if(subtitle)$
      \vspace{1em}
      {\large $subtitle$\par}
    $endif$

    $if(supervisor)$
      \vspace{2em}
      {\normalsize\itshape with the supervision of $supervisor$\par}
    $endif$

    $if(team)$
      \vspace{2em}
      {\large $for(team)$$team.id$ - $team.name$$sep$ \\ $endfor$\par}
    $endif$

    \vspace{3em}

    $if(university)$
      {\normalsize $university$\par}
    $endif$
    $if(course)$
      {\normalsize $course$\par}
    $endif$
    $if(academicyear)$
      {\normalsize $academicyear$\par}
    $endif$
  \end{minipage}
  \vfill
\end{titlepage}
}
